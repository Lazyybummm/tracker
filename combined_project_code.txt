const { contextBridge, ipcRenderer } = require('electron');

contextBridge.exposeInMainWorld('api', {
  // Auth methods
  saveAuth: (token, user) => ipcRenderer.invoke('save-auth', token, user),
  getAuth: () => ipcRenderer.invoke('get-auth'),
  clearAuth: () => ipcRenderer.invoke('clear-auth'),
  
  // Tracker methods
  checkIn: () => ipcRenderer.invoke('check-in'),
  checkOut: () => ipcRenderer.invoke('check-out'),
  getStatus: () => ipcRenderer.invoke('get-status'),
  
  // Events
  onScreenshotTaken: (callback) => {
    ipcRenderer.on('screenshot-taken', (event, filename) => callback(filename));
  },
  onStatusUpdate: (callback) => {
    ipcRenderer.on('status-update', (event, status) => callback(status));
  },
  onNextScreenshot: (callback) => {
    ipcRenderer.on('next-screenshot', (event, minutes) => callback(minutes));
  },
  onSessionStarted: (callback) => {
    ipcRenderer.on('session-started', () => callback());
  },
  
  // Helper to expose ipcRenderer for overlay
  getIpcRenderer: () => ipcRenderer
});

  const btnIcon           = document.getElementById('btnIcon');

  const statusDot         = document.getElementById('statusDot');

  let isCheckedIn = false;

  const ICON_PLAY = `<svg width="16" height="16" viewBox="0 0 16 16" fill="none"><polygon points="5,3 13,8 5,13" fill="currentColor"/></svg>`;
  const ICON_STOP = `<svg width="16" height="16" viewBox="0 0 16 16" fill="none"><rect x="4" y="4" width="8" height="8" rx="1.5" fill="currentColor"/></svg>`;
  const ICON_SPIN = `<svg width="16" height="16" viewBox="0 0 16 16" fill="none" style="animation:spin 0.8s linear infinite"><circle cx="8" cy="8" r="6" stroke="currentColor" stroke-width="2" stroke-dasharray="20 18"/></svg>`;

  function setBtn(state) {
    actionBtn.disabled = (state === 'loading-in' || state === 'loading-out');
    actionBtn.dataset.state = (state === 'stop') ? 'stop' : 'start';

    if (state === 'start') {
      btnIcon.innerHTML = ICON_PLAY;
      btnLabel.textContent = 'Start Session';
      actionBtn.dataset.state = 'start';
    } else if (state === 'stop') {
      btnIcon.innerHTML = ICON_STOP;
      btnLabel.textContent = 'End Session';
      actionBtn.dataset.state = 'stop';
    } else if (state === 'loading-in') {
      btnIcon.innerHTML = ICON_SPIN;
      btnLabel.textContent = 'Starting…';
    } else if (state === 'loading-out') {
      btnIcon.innerHTML = ICON_SPIN;
      btnLabel.textContent = 'Ending…';
    }
  }

  async function updateUIFromStatus() {
    try {
      const status = await window.api.getStatus();
      
      if (status.hasActiveSession) {
        isCheckedIn = true;
        setBtn('stop');
        statusCard.style.display = 'block';
        statusDot.className = 'status-dot active';
        statusText.textContent = 'Currently Working';
        if (headerStatusDot) headerStatusDot.className = 'live-dot online';
        if (headerStatusText) headerStatusText.textContent = 'ACTIVE';
        todayHoursElem.textContent = status.todayHours?.total_hours ?? '0.0';
        todaySessionsElem.textContent = status.todayHours?.session_count ?? '0';
        todayScreenshotsElem.textContent = status.todayHours?.total_screenshots ?? '0';
      } else {
        isCheckedIn = false;
        setBtn('start');
        statusCard.style.display = 'none';
        statusDot.className = 'status-dot';
        statusText.textContent = 'Not checked in';
        if (headerStatusDot) headerStatusDot.className = 'live-dot';
        if (headerStatusText) headerStatusText.textContent = 'READY';
      }
    } catch (err) {
      console.error('Status update error:', err);
      isCheckedIn = false;
      setBtn('start');
    }
  }

  async function checkIn() {
    setBtn('loading-in');
    try {
      const result = await window.api.checkIn();
      if (result.success) {
        await updateUIFromStatus();
      } else {
        console.error('Check-in failed:', result.error);
        setBtn('start');
      }
    } catch (err) {
      console.error('Check-in error:', err);
      setBtn('start');
    }
  }

  async function checkOut() {
    setBtn('loading-out');
    try {
      const result = await window.api.checkOut();
      if (result.success) {
        await updateUIFromStatus();
      } else {
        console.error('Check-out failed');
        setBtn('stop');
      }
    } catch (err) {
      console.error('Check-out error:', err);
      setBtn('stop');
    }
  }

  actionBtn.addEventListener('click', () => {
    if (isCheckedIn) checkOut();
    else checkIn();
  });

  window.api?.onStatusUpdate?.(() => updateUIFromStatus());
  
  // Initial load
  updateUIFromStatus();
}

  <script>
    const API_BASE = 'http://127.0.0.1:3000/api';
    
    // DOM Elements
    const loginModal = document.getElementById('loginModal');
    const loginBtn = document.getElementById('loginBtn');
    const loginIdentifier = document.getElementById('loginIdentifier');
    const loginPassword = document.getElementById('loginPassword');
    const loginError = document.getElementById('loginError');
    const userInfoBar = document.getElementById('userInfoBar');
    const userName = document.getElementById('userName');
    const userEmail = document.getElementById('userEmail');
    const logoutBtn = document.getElementById('logoutBtn');
    const actionBtn = document.getElementById('actionBtn');
    const statusCard = document.getElementById('statusCard');
    const statusDot = document.getElementById('statusDot');
    const statusText = document.getElementById('statusText');
    const todayHours = document.getElementById('todayHours');
    const todaySessions = document.getElementById('todaySessions');
    const todayScreenshots = document.getElementById('todayScreenshots');
    
    let authToken = null;
    let currentUser = null;
    let isActionInProgress = false;
    
    // Helper to get auth token - THIS IS THE ADDED FUNCTION
    window.getAuthToken = () => authToken;
    
    function showError(msg) {
      loginError.textContent = msg;
      loginError.classList.add('show');
      setTimeout(() => loginError.classList.remove('show'), 5000);
    }
    
    function hideLoginModal() {
      loginModal.classList.add('hidden');
    }
    
    function showAuthenticatedUI() {
      userInfoBar.classList.remove('hidden');
      actionBtn.classList.remove('hidden');
      userName.textContent = currentUser?.name || currentUser?.email || currentUser?.employee_code;
      userEmail.textContent = currentUser?.email || currentUser?.employee_code;
    }
    
    // Update status - ONLY gets status, does NOT trigger check-in/out
    async function updateStatus() {
      if (!authToken) return;
      
      try {
        const status = await window.api.getStatus();
        console.log('📊 Status update:', status);
        
        if (status.hasActiveSession) {
          statusCard.classList.add('visible');
          statusDot.classList.add('active');
          statusText.textContent = 'Currently Working';
          actionBtn.textContent = 'End Session';
        } else {
          statusCard.classList.remove('visible');
          statusDot.classList.remove('active');
          statusText.textContent = 'Not checked in';
          actionBtn.textContent = 'Start Session';
        }
        todayHours.textContent = status.todayHours?.total_hours || '0.0';
        todaySessions.textContent = status.todayHours?.session_count || '0';
        todayScreenshots.textContent = status.todayHours?.total_screenshots || '0';
      } catch (err) {
        console.error('Status update error:', err);
      }
    }
    
    // Check-in - ONLY called when user clicks Start Session
    async function checkIn() {
      if (isActionInProgress) {
        console.log('⏳ Action in progress, ignoring');
        return;
      }
      
      if (!authToken) {
        showError('Please login first');
        return;
      }
      
      isActionInProgress = true;
      actionBtn.disabled = true;
      actionBtn.textContent = 'Starting...';
      
      try {
        console.log('📞 Calling check-in via IPC');
        const data = await window.api.checkIn();
        console.log('📞 Check-in response:', data);
        
        if (data && data.success) {
          await updateStatus();
          console.log('✅ Check-in successful');
        } else {
          showError(data?.error || 'Check-in failed');
          await updateStatus();
        }
      } catch (err) {
        console.error('❌ Check-in error:', err);
        showError(err.message || 'Connection error');
      } finally {
        isActionInProgress = false;
        actionBtn.disabled = false;
      }
    }
    
    // Check-out - ONLY called when user clicks End Session
    async function checkOut() {
      if (isActionInProgress) {
        console.log('⏳ Action in progress, ignoring');
        return;
      }
      
      if (!authToken) return;
      
      isActionInProgress = true;
      actionBtn.disabled = true;
      actionBtn.textContent = 'Ending...';
      
      try {
        console.log('📞 Calling check-out via IPC');
        const data = await window.api.checkOut();
        console.log('📞 Check-out response:', data);
        
        if (data && data.success) {
          await updateStatus();
          console.log('✅ Check-out successful');
        } else {
          showError(data?.error || 'Check-out failed');
          await updateStatus();
        }
      } catch (err) {
        console.error('❌ Check-out error:', err);
        showError(err.message || 'Connection error');
      } finally {
        isActionInProgress = false;
        actionBtn.disabled = false;
      }
    }
    
    // Login handler
    async function handleLogin() {
      const username = loginIdentifier.value.trim();
      const password = loginPassword.value;
      
      if (!username || !password) {
        showError('Please enter email/employee code and password');
        return;
      }
      
      loginBtn.disabled = true;
      loginBtn.textContent = 'Signing in...';
      
      try {
        const response = await fetch(`${API_BASE}/auth/login`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ username, password })
        });
        
        const data = await response.json();
        
        if (data.success) {
          authToken = data.token;
          currentUser = data.user;
          
          if (window.api && window.api.saveAuth) {
            console.log('🔐 Saving auth to main process...');
            await window.api.saveAuth(data.token, currentUser);
          }
          
          localStorage.setItem('gridtrack_token', authToken);
          hideLoginModal();
          showAuthenticatedUI();
          await updateStatus();
          
          loginIdentifier.value = '';
          loginPassword.value = '';
        } else {
          showError(data.error || 'Login failed');
        }
      } catch (err) {
        console.error('Login error:', err);
        showError('Cannot connect to server');
      } finally {
        loginBtn.disabled = false;
        loginBtn.textContent = 'Sign In';
      }
    }
    
    // Logout handler
    async function handleLogout() {
      localStorage.removeItem('gridtrack_token');
      authToken = null;
      currentUser = null;
      
      if (window.api && window.api.clearAuth) {
        await window.api.clearAuth();
      }
      
      loginModal.classList.remove('hidden');
      userInfoBar.classList.add('hidden');
      actionBtn.classList.add('hidden');
      statusCard.classList.remove('visible');
    }
    
    // Check saved token on startup

        try {
          const response = await fetch(`${API_BASE}/auth/verify`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ token: savedToken })
          });
          const data = await response.json();
          
          if (data.success) {
            authToken = savedToken;
            currentUser = data.user;
            
            if (window.api && window.api.saveAuth) {
              await window.api.saveAuth(savedToken, currentUser);
            }
            
            hideLoginModal();
            showAuthenticatedUI();
            await updateStatus();
          } else {
            localStorage.removeItem('gridtrack_token');
          }
        } catch (err) {
          localStorage.removeItem('gridtrack_token');
        }
      }
    }
    
    // Enable paste in password field
    const passwordInput = document.getElementById('loginPassword');
    if (passwordInput) {

        e.stopPropagation();
        return true;

const screenshot = require('screenshot-desktop');
const sharp = require('sharp');

class ScreenshotService {
  async capture() {
    try {
      // Capture screenshot as buffer
      const imgBuffer = await screenshot({ format: 'png' });
      return imgBuffer;
    } catch (error) {
      console.error('Screenshot failed:', error);
      return null;
    }
  }
  
  async convertToJPEG(buffer, quality = 75) {
    try {
      const jpegBuffer = await sharp(buffer)
        .jpeg({ quality: quality, progressive: true, force: true })
        .toBuffer();
      return jpegBuffer;
    } catch (error) {
      console.error('JPEG conversion failed:', error);
      return buffer;
    }
  }
}

module.exports = new ScreenshotService();

const ALLOWED_ROLES = ['super_admin', 'admin', 'tenant_admin', 'org_admin'];

// ============= AUTHENTICATION =============

function saveAuth(token, user) {
  authToken = token;
  currentAdmin = user;
  localStorage.setItem('admin_token', token);
  localStorage.setItem('admin_user', JSON.stringify(user));
}

function clearAuth() {
  authToken = null;
  currentAdmin = null;
  localStorage.removeItem('admin_token');
  localStorage.removeItem('admin_user');
}

function loadSavedAuth() {
  const token = localStorage.getItem('admin_token');
  const user = localStorage.getItem('admin_user');
  if (token && user) {
    authToken = token;
    currentAdmin = JSON.parse(user);
    return true;
  }
  return false;
}

function getRoleDisplayName(role) {
  const roleNames = {
    'super_admin': 'Super Admin',
    'admin': 'Admin',
    'tenant_admin': 'Tenant Admin',
    'org_admin': 'Org Admin',
    'employee': 'Employee'
  };
  return roleNames[role] || role;
}

async function handleLogin() {
  const username = document.getElementById('loginUsername').value.trim();
  const password = document.getElementById('loginPassword').value;
  const loginBtn = document.getElementById('loginBtn');
  const loginError = document.getElementById('loginError');
  
  if (!username || !password) {
    loginError.textContent = 'Please enter email/employee code and password';
    loginError.classList.add('show');
    return;
  }
  
  loginBtn.disabled = true;
  loginBtn.textContent = 'Signing in...';
  
  try {
    const response = await fetch(`${API}/auth/login`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ username, password })
    });
    
    const data = await response.json();
    
    if (data.success) {
      // Allow all admin-level roles
      if (!ALLOWED_ROLES.includes(data.user.role)) {
        loginError.textContent = `Access denied. Your role "${data.user.role}" does not have permission to access this dashboard.`;
        loginError.classList.add('show');
        loginBtn.disabled = false;
        loginBtn.textContent = 'Sign In';
        return;
      }
      
      saveAuth(data.token, data.user);
      document.getElementById('loginOverlay').classList.add('hidden');
      document.getElementById('userInfo').style.display = 'flex';
      document.getElementById('userName').innerHTML = `${data.user.name || data.user.email} <span class="user-role-badge">${getRoleDisplayName(data.user.role)}</span>`;
      document.getElementById('userTenant').textContent = `Tenant ID: ${data.user.tenant_id || 'N/A'}${data.user.dept_id ? ` | Dept ID: ${data.user.dept_id}` : ''}`;
      document.getElementById('userAvatar').textContent = (data.user.name || data.user.email).substring(0, 1).toUpperCase();
      
      await loadDashboard();
      startAutoRefresh();
    } else {
      loginError.textContent = data.error || 'Login failed';
      loginError.classList.add('show');
    }
  } catch (err) {
    console.error('Login error:', err);
    loginError.textContent = 'Cannot connect to server';
    loginError.classList.add('show');
  } finally {
    loginBtn.disabled = false;
    loginBtn.textContent = 'Sign In';
  }
}

function handleLogout() {

  document.getElementById('userInfo').style.display = 'none';

  }
}

// ============= API CALLS WITH AUTH =============

async function apiCall(endpoint) { 
  try {
    const headers = {
      'Content-Type': 'application/json'
    };
    if (authToken) {
      headers['Authorization'] = `Bearer ${authToken}`;
    }
    const r = await fetch(API + endpoint, { headers }); 
    if (r.status === 401 || r.status === 403) {
      // Unauthorized - clear auth and show login
      clearAuth();
      document.getElementById('loginOverlay').classList.remove('hidden');
      document.getElementById('userInfo').style.display = 'none';
      showToast('Session expired. Please login again.', true);
      throw new Error('Unauthorized');
    }
    if (!r.ok) throw new Error('HTTP ' + r.status); 
    return await r.json(); 
  } catch (error) {
    console.error('API Error:', error);
    throw error;
  }
}

async function checkBackendAndLoad() {
  try {
    const response = await fetch('http://127.0.0.1:3000/api/health');
    if (response.ok) {
      isBackendDown = false;
      setApiStatus(true);
      await loadDashboard();
      startAutoRefresh();
    } else {
      throw new Error('Backend not ready');
    }
  } catch (error) {
    isBackendDown = true;
    setApiStatus(false);
    showBackendOfflineMessage();
  }
}

async function loadReports() {
  if (isBackendDown || !authToken) return;
  try { 
    const data = await apiCall('/admin/users'), emps = data.users || []; 
    const ctx = document.getElementById('monthlyChart').getContext('2d'); 
    if (monthChart) monthChart.destroy(); 
    const buckets = {'0–10h':0,'10–20h':0,'20–40h':0,'40+h':0}; 
    emps.forEach(e => { const h = e.totalWorkingHours||0; if (h<10) buckets['0–10h']++; else if (h<20) buckets['10–20h']++; else if (h<40) buckets['20–40h']++; else buckets['40+h']++; }); 
    monthChart = new Chart(ctx, { 
      type: 'doughnut', 
      data: { 
        labels: Object.keys(buckets), 
        datasets: [{ 
          data: Object.values(buckets), 
          backgroundColor: ['#9fe870','#ffc091','#38c8ff','#d03238'], 
          borderColor: document.documentElement.getAttribute('data-theme') === 'dark' ? '#1a1b18' : '#e8ebe6', 
          borderWidth: 3 
        }] 
      }, 
      options: { 
        responsive: true, 
        maintainAspectRatio: true, 
        cutout: '65%', 
        plugins: { 
          legend: { position: 'bottom', labels: { color: '#868685', font: { size: 11 } } } 
        } 
      } 
    }); 
    const active = emps.filter(e => e.hasActiveSession).length, 
          totalH = emps.reduce((s,e)=> s+(e.totalWorkingHours||0),0), 
          avgH = emps.length ? (totalH / emps.length).toFixed(1) : 0; 
    document.getElementById('reportsSummary').innerHTML = [
      ['Total Employees', emps.length],
      ['Active Now', active],
      ['Total Hours Logged', totalH.toFixed(1)+'h'],
      ['Avg Hours / Employee', avgH+'h'],
      ['Total Captures', emps.reduce((s,e)=> s+(e.screenshotCount||0),0)]
    ].map(([l,v]) => `

// Helper function to make API calls with authentication
async function apiCall(endpoint, options = {}) {
  const url = `${backendBaseUrl}${endpoint}`;
  console.log(`🌐 API Call: ${options.method || 'GET'} ${url}`);
  
  try {
    const headers = {
      'Content-Type': 'application/json',
      ...options.headers
    };
    
    if (authToken) {
      headers['Authorization'] = `Bearer ${authToken}`;
    }
    
    const response = await fetch(url, {
      ...options,
      headers
    });
    
    const contentType = response.headers.get('content-type');
    if (!contentType || !contentType.includes('application/json')) {
      const text = await response.text();
      console.error(`❌ Non-JSON response from ${endpoint}:`, text.substring(0, 200));
      throw new Error(`Server returned ${contentType || 'unknown content type'}. Expected JSON.`);
    }
    
    const data = await response.json();
    
    if (!response.ok) {
      throw new Error(data.error || `HTTP ${response.status}`);
    }
    
    return data;
  } catch (error) {
    console.error(`❌ API call failed (${endpoint}):`, error.message);
    throw error;
  }
}

  <div class="notif-stack" id="notificationContainer"></div>

  <script>
    const timerEl = document.getElementById('sessionTimer');
    let startTs = null;
    let rafId = null;
    let running = false;

    function pad(n) { return String(n).padStart(2, '0'); }

    function formatElapsed(ms) {
      const totalSec = Math.floor(ms / 1000);
      const h = Math.floor(totalSec / 3600);
      const m = Math.floor((totalSec % 3600) / 60);
      const s = totalSec % 60;
      return h > 0 ? `${pad(h)}:${pad(m)}:${pad(s)}` : `${pad(m)}:${pad(s)}`;
    }

    function startTimer() {
      if (running) stopTimer();
      startTs = Date.now();
      running = true;
      
      function tick() {
        if (!running) return;
        timerEl.textContent = formatElapsed(Date.now() - startTs);
        rafId = requestAnimationFrame(tick);
      }
      tick();
    }

    function stopTimer() {
      running = false;
      if (rafId) cancelAnimationFrame(rafId);
      timerEl.textContent = '00:00';
    }

    // Drag functionality
    const pill = document.getElementById('pill');
    let dragging = false, ox = 0, oy = 0, startX = 0, startY = 0;

    pill.addEventListener('mousedown', e => {
      if (e.target.closest('#checkoutBtn')) return;
      dragging = true;
      startX = e.clientX;
      startY = e.clientY;
      const r = pill.getBoundingClientRect();
      ox = r.left;
      oy = r.top;
      e.preventDefault();
    });

    document.addEventListener('mousemove', e => {
      if (!dragging) return;
      const nx = Math.max(8, Math.min(ox + e.clientX - startX, window.innerWidth - pill.offsetWidth - 8));
      const ny = Math.max(8, Math.min(oy + e.clientY - startY, window.innerHeight - pill.offsetHeight - 8));
      pill.parentElement.style.cssText = `position:fixed;left:${nx}px;top:${ny}px;bottom:auto`;
    });

    document.addEventListener('mouseup', () => { dragging = false; });

    // Event handlers using window.api
    if (window.api) {
      // Session started event
      window.api.onSessionStarted(() => {
        startTimer();
      });

      // Screenshot notification
      window.api.onScreenshotTaken((filename) => {
        const container = document.getElementById('notificationContainer');
        const n = document.createElement('div');
        n.className = 'notification';
        n.innerHTML = `<svg width="13" height="13" viewBox="0 0 14 14" fill="none"><rect x="1.5" y="3" width="11" height="8.5" rx="1.5" stroke="#C9922A" stroke-width="1.2"/><circle cx="7" cy="7.2" r="2" stroke="#C9922A" stroke-width="1.2"/></svg><span>${filename}</span>`;
        container.appendChild(n);
        setTimeout(() => {
          n.style.opacity = '0';
          n.style.transform = 'translateX(12px)';
          setTimeout(() => n.remove(), 280);
        }, 3200);
      });

      // Next screenshot notification
      window.api.onNextScreenshot((minutes) => {
        const container = document.getElementById('notificationContainer');
        const n = document.createElement('div');
        n.className = 'notification';
        n.innerHTML = `<svg width="13" height="13" viewBox="0 0 14 14" fill="none"><circle cx="7" cy="7" r="5.5" stroke="#C9922A" stroke-width="1.3"/><path d="M7 4.5v2.8l1.8 1.2" stroke="#C9922A" stroke-width="1.3" stroke-linecap="round" stroke-linejoin="round"/></svg><span>Next capture in ~${minutes} min</span>`;
        container.appendChild(n);
        setTimeout(() => {
          n.style.opacity = '0';
          n.style.transform = 'translateX(12px)';
          setTimeout(() => n.remove(), 2500);
        }, 2500);
      });

      // Checkout button
      document.getElementById('checkoutBtn').addEventListener('click', async () => {
        stopTimer();
        if (window.api.checkOut) await window.api.checkOut();
      });

      // Auto-start timer if session already active
      window.api.getStatus().then(status => {
        if (status && status.hasActiveSession) startTimer();
      }).catch(err => console.log('Could not get initial status:', err));
    }
  </script>
