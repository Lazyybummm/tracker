const { contextBridge, ipcRenderer } = require('electron');

// Expose protected methods that allow the renderer process to use
// the ipcRenderer without exposing the entire object
contextBridge.exposeInMainWorld('api', {
  checkIn: (phone, name) => ipcRenderer.invoke('check-in', phone, name),
  checkOut: () => ipcRenderer.invoke('check-out'),
  getStatus: () => ipcRenderer.invoke('get-status'),
  saveCredentials: (phone, name) => ipcRenderer.invoke('save-credentials', phone, name),
  getCredentials: () => ipcRenderer.invoke('get-credentials'),
  saveOverlayPosition: (bounds) => ipcRenderer.invoke('save-overlay-position', bounds),
  getOverlayPosition: () => ipcRenderer.invoke('get-overlay-position'),
  onScreenshotTaken: (callback) => {
    ipcRenderer.on('screenshot-taken', (event, filename) => callback(filename));
  },
  onStatusUpdate: (callback) => {
    ipcRenderer.on('status-update', (event, status) => callback(status));
  },
  onNextScreenshot: (callback) => {
    ipcRenderer.on('next-screenshot', (event, minutes) => callback(minutes));
  }
});

console.log('Renderer loaded');

// Check if we're in the overlay window
const isOverlay = window.location.pathname.includes('overlay.html');

if (isOverlay) {
  // Overlay window logic
  const checkoutBtn = document.getElementById('checkoutBtn');
  const timerValue = document.getElementById('timerValue');
  let nextIntervalMinutes = null;
  
  // Listen for next screenshot timing
  if (window.api.onScreenshotTaken) {
    window.api.onScreenshotTaken((filename) => {
      const container = document.getElementById('notification-container');
      const notification = document.createElement('div');
      notification.className = 'notification';
      notification.innerHTML = `📸 Screenshot captured: ${filename}`;
      container.appendChild(notification);
      
      setTimeout(() => {
        notification.remove();
      }, 3000);
    });
  }
  
  // Listen for next screenshot timing updates
  if (window.api.onStatusUpdate) {
    window.api.onStatusUpdate((data) => {
      // Handle status updates if needed
    });
  }
  
  // Custom event listener for next screenshot timing
  window.electronAPI = window.api;
  
  checkoutBtn.addEventListener('click', async () => {
    await window.api.checkOut();
  });
  
} else {
  // Check-in window logic
  const actionBtn = document.getElementById('actionBtn');
  const nameInput = document.getElementById('nameInput');
  const phoneInput = document.getElementById('phoneInput');
  const statusCard = document.getElementById('statusCard');
  const statusIndicator = document.getElementById('statusIndicator');
  const statusText = document.getElementById('statusText');
  const todayHoursElem = document.getElementById('todayHours');
  const todaySessionsElem = document.getElementById('todaySessions');
  const todayScreenshotsElem = document.getElementById('todayScreenshots');
  
  let isCheckedIn = false;
  
  // Load saved credentials
  async function loadSavedCredentials() {
    const credentials = await window.api.getCredentials();
    if (credentials.phone) {
      phoneInput.value = credentials.phone;
    }
    if (credentials.name) {
      nameInput.value = credentials.name;
    }
  }
  
  // Update UI based on check-in status
  async function updateStatus() {
    const status = await window.api.getStatus();
    
    if (status.hasActiveSession) {
      isCheckedIn = true;
      actionBtn.textContent = 'Stop Working';
      actionBtn.classList.remove('btn-primary');
      actionBtn.classList.add('btn-danger');
      statusCard.style.display = 'block';
      statusIndicator.className = 'status-indicator active';
      statusText.textContent = 'Currently Working';
      todayHoursElem.textContent = status.todayHours?.total_hours || '0';
      todaySessionsElem.textContent = status.todayHours?.session_count || '0';
      todayScreenshotsElem.textContent = status.todayHours?.total_screenshots || '0';
      
      // Disable input fields
      nameInput.disabled = true;
      phoneInput.disabled = true;
    } else {
      isCheckedIn = false;
      actionBtn.textContent = 'Start Working';
      actionBtn.classList.remove('btn-danger');
      actionBtn.classList.add('btn-primary');
      statusCard.style.display = 'none';
      
      // Enable input fields
      nameInput.disabled = false;
      phoneInput.disabled = false;
    }
  }
  
  // Check in
  async function checkIn() {
    const name = nameInput.value.trim();
    const phone = phoneInput.value.trim();
    
    if (!name || !phone) {
      alert('Please enter both name and phone number');
      return;
    }
    
    actionBtn.disabled = true;
    actionBtn.textContent = 'Checking in...';
    
    const result = await window.api.checkIn(phone, name);
    
    if (result.success) {
      console.log('Check-in successful');
      await updateStatus();
    } else {
      alert('Check-in failed: ' + (result.error || 'Unknown error'));
      actionBtn.disabled = false;
      actionBtn.textContent = 'Start Working';
    }
  }
  
  // Check out
  async function checkOut() {
    actionBtn.disabled = true;
    actionBtn.textContent = 'Checking out...';
    
    const result = await window.api.checkOut();
    
    if (result.success) {
      console.log('Check-out successful');
      await updateStatus();
    } else {
      alert('Check-out failed');
      actionBtn.disabled = false;
      actionBtn.textContent = 'Stop Working';
    }
  }
  
  // Handle button click
  actionBtn.addEventListener('click', async () => {
    if (isCheckedIn) {
      await checkOut();
    } else {
      await checkIn();
    }
  });
  
  // Load saved credentials and update status
  loadSavedCredentials();
  updateStatus();
  
  // Listen for status updates from main process
  if (window.api.onStatusUpdate) {
    window.api.onStatusUpdate(() => {
      updateStatus();
    });
  }
}

const screenshot = require('screenshot-desktop');
const sharp = require('sharp');

class ScreenshotService {
  async capture() {
    try {
      // Capture screenshot as buffer
      const imgBuffer = await screenshot();
      return imgBuffer;
    } catch (error) {
      console.error('Screenshot failed:', error);
      return null;
    }
  }
  
  async convertToJPEG(buffer, quality = 75) {
    try {
      const jpegBuffer = await sharp(buffer)
        .jpeg({ quality: quality, progressive: true })
        .toBuffer();
      return jpegBuffer;
    } catch (error) {
      console.error('JPEG conversion failed:', error);
      return buffer;
    }
  }
}

module.exports = new ScreenshotService();

  const theme = saved || (prefersDark ? 'dark' : 'light');
  setTheme(theme);

async function loadDashboard() {
  try {
    const data = await api('/admin/users');
    allEmployees = data.users || [];
    setApiStatus(true);
    tick();

    const active = allEmployees.filter(e => e.hasActiveSession).length;
    const hrs = allEmployees.reduce((s, e) => s + (e.todayHours || 0), 0);
    const caps = allEmployees.reduce((s, e) => s + (e.screenshotCount || 0), 0);

    animVal('totalEmployees', allEmployees.length);
    animVal('activeEmployees', active);
    document.getElementById('totalHours').textContent = hrs.toFixed(1);
    animVal('totalScreenshots', caps);

    buildActivityChart(allEmployees);
    buildRecentList(allEmployees);
  } catch {
    setApiStatus(false);
  }
}

function animVal(id, target) {
  const el = document.getElementById(id);
  const start = parseInt(el.textContent) || 0;
  const dur = 600;
  const t0 = performance.now();
  const fn = now => {
    const p = Math.min((now - t0) / dur, 1);
    el.textContent = Math.round(start + (target - start) * (p < 1 ? 1 - Math.pow(1 - p, 3) : 1));
    if (p < 1) requestAnimationFrame(fn);
  };
  requestAnimationFrame(fn);
}

async function loadReports() {
  try {
    const data = await api('/admin/users');
    const emps = data.users || [];

    const ctx = document.getElementById('monthlyChart').getContext('2d');
    if (monthChart) monthChart.destroy();

    const buckets = {'0–10h':0,'10–20h':0,'20–40h':0,'40+h':0};
    emps.forEach(e => {
      const h = e.totalWorkingHours||0;
      if (h < 10) buckets['0–10h']++;
      else if (h < 20) buckets['10–20h']++;
      else if (h < 40) buckets['20–40h']++;
      else buckets['40+h']++;
    });

    const isDark = document.documentElement.getAttribute('data-theme') === 'dark';
    monthChart = new Chart(ctx, {
      type: 'doughnut',
      data: {
        labels: Object.keys(buckets),
        datasets: [{
          data: Object.values(buckets),
          backgroundColor: ['rgba(0,212,180,0.8)','rgba(245,158,11,0.8)','rgba(129,140,248,0.8)','rgba(244,63,94,0.8)'],
          borderColor: [isDark ? '#0d0f12' : '#ffffff'],
          borderWidth: 3,
          hoverOffset: 8
        }]
      },
      options: {
        responsive: true,
        maintainAspectRatio: true,
        cutout: '65%',
        plugins: {
          legend: {
            position: 'bottom',
            labels: { color: isDark ? '#8B93A7' : '#475569', font: { family: "'DM Mono', monospace", size: 11 }, padding: 16 }
          },
          tooltip: {
            backgroundColor: isDark ? '#1a2030' : '#ffffff',
            borderColor: 'rgba(0,212,180,0.3)',
            borderWidth: 1,
            titleColor: isDark ? '#F0EDE6' : '#0F172A',
            bodyColor: isDark ? '#8B93A7' : '#475569',
            titleFont: { family: "'DM Mono', monospace" },
            bodyFont: { family: "'DM Mono', monospace" }
          }
        }
      }
    });

    const active = emps.filter(e => e.hasActiveSession).length;
    const totalH = emps.reduce((s,e) => s+(e.totalWorkingHours||0), 0);
    const avgH = emps.length ? (totalH / emps.length).toFixed(1) : 0;
    document.getElementById('reportsSummary').innerHTML = [
      ['Total Employees', emps.length],
      ['Active Now', active],
      ['Total Hours Logged', totalH.toFixed(1) + 'h'],
      ['Avg Hours / Employee', avgH + 'h'],
      ['Total Captures', emps.reduce((s,e) => s+(e.screenshotCount||0),0)],
    ].map(([l,v]) => `
      <div style="display:flex;justify-content:space-between;align-items:center;padding:13px 0;border-bottom:1px solid var(--border-faint);">
        <span style="font-family:var(--font-mono);font-size:11px;color:var(--text-lo);letter-spacing:.08em;text-transform:uppercase">${l}</span>
        <span style="font-family:var(--font-display);font-size:22px;letter-spacing:.02em;color:var(--text-hi)">${v}</span>
      </div>
    `).join('') + `<div style="padding:13px 0;text-align:center;font-family:var(--font-mono);font-size:9px;color:var(--text-lo);letter-spacing:.1em">LIVE DATA · AUTO-REFRESH 30s</div>`;
  } catch {
    console.error('Reports failed');
  }
}

// Helper function to make API calls with error handling
async function apiCall(endpoint, options = {}) {
  const url = `${backendBaseUrl}${endpoint}`;
  console.log(`🌐 API Call: ${options.method || 'GET'} ${url}`);
  
  try {
    const response = await fetch(url, {
      ...options,
      headers: {
        'Content-Type': 'application/json',
        ...options.headers
      }
    });
    
    // Check if response is JSON
    const contentType = response.headers.get('content-type');
    if (!contentType || !contentType.includes('application/json')) {
      const text = await response.text();
      console.error(`❌ Non-JSON response from ${endpoint}:`, text.substring(0, 200));
      throw new Error(`Server returned ${contentType || 'unknown content type'}. Expected JSON.`);
    }
    
    const data = await response.json();
    
    if (!response.ok) {
      throw new Error(data.error || `HTTP ${response.status}`);
    }
    
    return data;
  } catch (error) {
    console.error(`❌ API call failed (${endpoint}):`, error.message);
    throw error;
  }
}

      window.api.onScreenshotTaken((filename) => {
        const container = document.getElementById('notificationContainer');
        const notification = document.createElement('div');
        notification.className = 'notification';
        notification.innerHTML = `
          <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="#D4A843" stroke-width="2">
            <rect x="2" y="4" width="20" height="16" rx="2" stroke="currentColor"/>
            <circle cx="12" cy="12" r="3" stroke="currentColor"/>
          </svg>
          <span>${filename}</span>
        `;
        container.appendChild(notification);
        
        setTimeout(() => {
          notification.style.animation = 'slideInRight 0.3s reverse';
          setTimeout(() => notification.remove(), 300);
        }, 3000);
      });
    }
    
    // Handle checkout - stop timer when exiting
    const checkoutBtn = document.getElementById('checkoutBtn');
    if (checkoutBtn) {
      checkoutBtn.addEventListener('click', async () => {
        stopTimer();
        if (window.api && window.api.checkOut) {
          await window.api.checkOut();
        }
      });
    }
    
    // Expose control functions for debugging
    window.debugTimer = {
      start: startTimer,
      stop: stopTimer,
      isRunning: () => isTimerRunning,
      getElapsed: () => sessionStartTime ? Math.floor((Date.now() - sessionStartTime) / 1000) : 0
    };
    
    console.log('✅ Overlay loaded - timer ready');
  </script>
</body>
